# minuto hora dia mes dia-semana  comando
#
# Logistica primero, ventas despues. Van escalonados de 15 minutos
# para no abrir cuatro sesiones simultaneas contra el SGE: la de
# ventas devuelve mas de 100 mil filas y tarda cerca de dos minutos.
#
# A las 5 AM la operacion de "ayer" ya suele estar cerrada.

0  5 * * * cd /app && /usr/local/bin/python /app/etl.py --empresa cfs >> /var/log/etl.log 2>&1
15 5 * * * cd /app && /usr/local/bin/python /app/etl.py --empresa acabados >> /var/log/etl.log 2>&1
30 5 * * * cd /app && /usr/local/bin/python /app/etl_ventas.py --empresa cfs >> /var/log/etl.log 2>&1
45 5 * * * cd /app && /usr/local/bin/python /app/etl_ventas.py --empresa acabados >> /var/log/etl.log 2>&1
